// ============================================================
// GOOGLE APPS SCRIPT DATABASE - ARISAN & KAS RT 39 DESA BELOTAN
// ============================================================

function doGet(e) {
  var action = e && e.parameter ? e.parameter.action : '';
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  
  if (action === 'getData') {
    var result = readAllDatabase(ss);
    return ContentService.createTextOutput(JSON.stringify({
      status: 'success',
      data: result
    })).setMimeType(ContentService.MimeType.JSON);
  }
  
  return ContentService.createTextOutput(JSON.stringify({
    status: 'success',
    message: 'API Database Arisan & Kas RT 39 Aktif dan Siap Digunakan.'
  })).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    var raw = e.postData.contents;
    var payload = JSON.parse(raw);
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    
    writeAllDatabase(ss, payload);
    
    return ContentService.createTextOutput(JSON.stringify({
      status: 'success',
      message: 'Data Arisan & Kas RT 39 berhasil disimpan ke Google Sheet.',
      timestamp: new Date().toISOString()
    })).setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({
      status: 'error',
      message: err.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  }
}

function writeAllDatabase(ss, data) {
  // 1. Sheet Data_Arisan
  var shArisan = getOrCreateSheet(ss, 'Data_Arisan', ['ID', 'Nama', 'Slot Ke', 'Display Nama', 'Status Bayar', 'Sudah Menang']);
  clearDataRows(shArisan);
  if (data.anggotaArisan && data.anggotaArisan.length > 0) {
    var rowsArisan = data.anggotaArisan.map(function(a) {
      return [a.id, a.nama, a.slotKe, a.displayNama, a.bayarArisan ? 'Lunas' : 'Belum', a.sudahMenang ? 'Menang' : 'Belum'];
    });
    shArisan.getRange(2, 1, rowsArisan.length, 6).setValues(rowsArisan);
  }
  
  // 2. Sheet Kas_Warga
  var shKas = getOrCreateSheet(ss, 'Kas_Warga', ['ID', 'Nama Warga', 'Alamat', 'Status Bayar Kas']);
  clearDataRows(shKas);
  if (data.wargaKas && data.wargaKas.length > 0) {
    var rowsKas = data.wargaKas.map(function(w) {
      return [w.id, w.nama, w.alamat || '', w.bayarKas ? 'Lunas' : 'Belum'];
    });
    shKas.getRange(2, 1, rowsKas.length, 4).setValues(rowsKas);
  }
  
  // 3. Sheet Transaksi_Buku_Kas
  var shTrans = getOrCreateSheet(ss, 'Buku_Kas', ['ID', 'Tanggal', 'Tipe', 'Keterangan', 'Nominal']);
  clearDataRows(shTrans);
  if (data.transaksiLain && data.transaksiLain.length > 0) {
    var rowsTrans = data.transaksiLain.map(function(t) {
      return [t.id, t.tanggal, t.tipe, t.keterangan, t.nominal];
    });
    shTrans.getRange(2, 1, rowsTrans.length, 5).setValues(rowsTrans);
  }

  // 4. Sheet Riwayat_Pemenang
  var shMenang = getOrCreateSheet(ss, 'Riwayat_Pemenang', ['ID', 'Nama Pemenang', 'Slot', 'Tanggal Menang']);
  clearDataRows(shMenang);
  if (data.riwayatPemenang && data.riwayatPemenang.length > 0) {
    var rowsMenang = data.riwayatPemenang.map(function(r) {
      return [r.id, r.nama, r.displayNama, r.tanggal || ''];
    });
    shMenang.getRange(2, 1, rowsMenang.length, 4).setValues(rowsMenang);
  }

  // 5. Sheet Pengaturan_Sistem
  var shSetting = getOrCreateSheet(ss, 'Pengaturan', ['Parameter', 'Nilai']);
  clearDataRows(shSetting);
  var rowsSet = [
    ['Iuran Arisan per Slot', data.iuranArisan || 10000],
    ['Iuran Kas RT per Warga', data.iuranKas || 5000],
    ['Password Admin', data.adminPassword || 'rt39'],
    ['Terakhir Diperbarui', new Date().toLocaleString('id-ID')]
  ];
  shSetting.getRange(2, 1, rowsSet.length, 2).setValues(rowsSet);
}

function readAllDatabase(ss) {
  var result = {
    anggotaArisan: [],
    wargaKas: [],
    transaksiLain: [],
    riwayatPemenang: [],
    iuranArisan: 10000,
    iuranKas: 5000
  };

  // Baca Arisan
  var shArisan = ss.getSheetByName('Data_Arisan');
  if (shArisan && shArisan.getLastRow() > 1) {
    var vals = shArisan.getRange(2, 1, shArisan.getLastRow() - 1, 6).getValues();
    result.anggotaArisan = vals.map(function(r) {
      return {
        id: Number(r[0]),
        nama: String(r[1]),
        slotKe: Number(r[2]),
        displayNama: String(r[3]),
        bayarArisan: String(r[4]).toLowerCase() === 'lunas',
        sudahMenang: String(r[5]).toLowerCase() === 'menang'
      };
    });
  }

  // Baca Kas
  var shKas = ss.getSheetByName('Kas_Warga');
  if (shKas && shKas.getLastRow() > 1) {
    var valsKas = shKas.getRange(2, 1, shKas.getLastRow() - 1, 4).getValues();
    result.wargaKas = valsKas.map(function(r) {
      return {
        id: Number(r[0]),
        nama: String(r[1]),
        alamat: String(r[2]),
        bayarKas: String(r[3]).toLowerCase() === 'lunas'
      };
    });
  }

  // Baca Transaksi
  var shTrans = ss.getSheetByName('Buku_Kas');
  if (shTrans && shTrans.getLastRow() > 1) {
    var valsT = shTrans.getRange(2, 1, shTrans.getLastRow() - 1, 5).getValues();
    result.transaksiLain = valsT.map(function(r) {
      return {
        id: Number(r[0]),
        tanggal: String(r[1]),
        tipe: String(r[2]),
        keterangan: String(r[3]),
        nominal: Number(r[4])
      };
    });
  }

  // Baca Riwayat
  var shMenang = ss.getSheetByName('Riwayat_Pemenang');
  if (shMenang && shMenang.getLastRow() > 1) {
    var valsM = shMenang.getRange(2, 1, shMenang.getLastRow() - 1, 4).getValues();
    result.riwayatPemenang = valsM.map(function(r) {
      return {
        id: Number(r[0]),
        nama: String(r[1]),
        displayNama: String(r[2]),
        tanggal: String(r[3])
      };
    });
  }

  // Baca Setting
  var shSet = ss.getSheetByName('Pengaturan');
  if (shSet && shSet.getLastRow() > 1) {
    var valsS = shSet.getRange(2, 1, shSet.getLastRow() - 1, 2).getValues();
    valsS.forEach(function(r) {
      if (String(r[0]).indexOf('Iuran Arisan') !== -1) result.iuranArisan = Number(r[1]);
      if (String(r[0]).indexOf('Iuran Kas') !== -1) result.iuranKas = Number(r[1]);
      if (String(r[0]).indexOf('Password Admin') !== -1 && r[1]) result.adminPassword = String(r[1]).trim();
    });
  }

  return result;
}

function getOrCreateSheet(ss, name, headers) {
  var sheet = ss.getSheetByName(name);
  if (!sheet) {
    sheet = ss.insertSheet(name);
    sheet.getRange(1, 1, 1, headers.length).setValues([headers]);
    sheet.getRange(1, 1, 1, headers.length).setFontWeight('bold').setBackground('#e2e8f0');
  }
  return sheet;
}

function clearDataRows(sheet) {
  if (sheet.getLastRow() > 1) {
    sheet.getRange(2, 1, sheet.getLastRow() - 1, sheet.getLastColumn()).clearContent();
  }
}

Panduan Deployment Google Apps Script

  1. Buka Google Sheets Baru di browser Anda, beri nama spreadsheet misalnya: Database Arisan RT 39.
  2. Di menu atas Google Sheet, klik Ekstensi (Extensions) > Apps Script.
  3. Hapus kode bawaan di editor, lalu klik tombol Salin Kode Script di atas dan tempelkan ke editor Apps Script.
  4. Klik tombol icon Simpan (Ctrl+S) pada Apps Script.
  5. Klik tombol biru Terapkan (Deploy) > Deployment baru (New deployment) di pojok kanan atas.
  6. Pada icon gerigi (Select type), pilih Aplikasi web (Web app).
  7. Isi deskripsi: Arisan RT 39 Web API.
  8. Pada bagian Jalankan sebagai (Execute as): pilih Saya (email Anda).
  9. Pada bagian Yang memiliki akses (Who has access): pilih Siapa saja (Anyone) (Wajib pilih ini agar data bisa dikirim tanpa login ulang).
  10. Klik Deploy, lalu klik Beri Akses (Authorize access) jika diminta konfirmasi akun Google.
  11. Salin URL Aplikasi Web yang berakhiran /exec, lalu tempelkan ke kolom input URL di atas dan klik Simpan & Tes Koneksi!
